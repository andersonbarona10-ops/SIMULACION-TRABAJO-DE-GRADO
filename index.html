import math import matplotlib.pyplot as plt # =============================== # PARAMETROS DEL MODELO # =============================== RACK_SPACING = 1.2 # m AISLE_WIDTH = 2.5 # m NUM_RACKS = 30 WALK_SPEED = 1.0 # m/s PICK_TIME = 15 # s # Escenario utilizado en la monografía picks_p1 = [30,27,24,21,18,15,12,9,6,3] picks_p2 = [28,26,22,20,17,14,11,8,5,2] # =============================== # COORDENADAS # =============================== def rack_coord(aisle, rack): x = rack * RACK_SPACING if aisle == 1: y = 0 else: y = AISLE_WIDTH return (x,y) # =============================== # DISTANCIA EUCLIDIANA # =============================== def distance(p1,p2): return math.sqrt( (p2[0]-p1[0])**2 + (p2[1]-p1[1])**2 ) # =============================== # METODO ACTUAL # =============================== def simulate_current(): total_distance = 0 route_x = [0] route_y = [0] # PASILLO 1 farthest = max(picks_p1) total_distance += farthest * RACK_SPACING total_distance += farthest * RACK_SPACING route_x.extend([ rack_coord(1,farthest)[0], 0 ]) route_y.extend([ rack_coord(1,farthest)[1], 0 ]) # PASILLO 2 total_distance += AISLE_WIDTH route_x.append(0) route_y.append(AISLE_WIDTH) farthest = max(picks_p2) total_distance += farthest * RACK_SPACING total_distance += farthest * RACK_SPACING route_x.extend([ rack_coord(2,farthest)[0], 0 ]) route_y.extend([ rack_coord(2,farthest)[1], AISLE_WIDTH ]) return total_distance, route_x, route_y # =============================== # METODO RI-RSP # =============================== def simulate_rirsp(): route = [] route.append(rack_coord(1,30)) sequence = [ (1,30),(2,28), (1,27),(2,26), (1,24),(2,22), (1,21),(2,20), (1,18),(2,17), (1,15),(2,14), (1,12),(2,11), (1,9),(2,8), (1,6),(2,5), (1,3),(2,2) ] total_distance = 30 * RACK_SPACING for i in range(len(sequence)-1): p1 = rack_coord( sequence[i][0], sequence[i][1] ) p2 = rack_coord( sequence[i+1][0], sequence[i+1][1] ) total_distance += distance(p1,p2) total_distance += 2 * RACK_SPACING route_x = [] route_y = [] route_x.append(0) route_y.append(0) route_x.append( rack_coord(1,30)[0] ) route_y.append( rack_coord(1,30)[1] ) for a,r in sequence: route_x.append( rack_coord(a,r)[0] ) route_y.append( rack_coord(a,r)[1] ) route_x.append(0) route_y.append(0) return total_distance, route_x, route_y # =============================== # EJECUCION # =============================== dist_actual, x1, y1 = simulate_current() dist_rirsp, x2, y2 = simulate_rirsp() time_actual = dist_actual / WALK_SPEED time_rirsp = dist_rirsp / WALK_SPEED pick_time_total = ( 20 * PICK_TIME ) cycle_actual = ( time_actual + pick_time_total ) cycle_rirsp = ( time_rirsp + pick_time_total ) improvement = ( ( dist_actual - dist_rirsp ) / dist_actual ) * 100 # =============================== # RESULTADOS # =============================== print("\n======= RESULTADOS =======") print( f"Distancia actual: {dist_actual:.2f} m" ) print( f"Distancia RI-RSP: {dist_rirsp:.2f} m" ) print( f"Mejora: {improvement:.2f}%" ) print( f"Tiempo ciclo actual: {cycle_actual/60:.2f} min" ) print( f"Tiempo ciclo RI-RSP: {cycle_rirsp/60:.2f} min" ) # =============================== # GRAFICA RUTAS # =============================== plt.figure(figsize=(10,6)) plt.plot( x1,y1, marker='o', linewidth=2, label="Método Actual" ) plt.plot( x2,y2, marker='s', linewidth=2, label="RI-RSP" ) plt.xlabel("Longitud almacén (m)") plt.ylabel("Pasillo") plt.title( "Comparación de rutas de picking" ) plt.legend() plt.grid(True) plt.show() # =============================== # GRAFICA KPI # =============================== plt.figure(figsize=(6,5)) metodos = [ "Actual", "RI-RSP" ] distancias = [ dist_actual, dist_rirsp ] plt.bar( metodos, distancias ) plt.ylabel("Distancia (m)") plt.title( "Distancia recorrida" ) plt.show()
